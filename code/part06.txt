    float entryPrice =

    if strategy.position_size[1] <= 0
        longStop :=
             entryPrice - stopDistance

        longTarget :=
             entryPrice + targetDistance


    // Break-even
    if useBreakEven and
       high - entryPrice >= breakEvenStart

        float beStop =
             entryPrice + breakEvenExtra

        longStop :=
             na(longStop) ?
             beStop :
             math.max(longStop, beStop)


    // Trailing
    if useTrailing and
       high - entryPrice >= trailStart

        float trailingCandidate =
             high - trailDistance

        longStop :=
             na(longStop) ?
             trailingCandidate :
             math.max(
                 longStop,
                 trailingCandidate
             )


    strategy.exit(
         "LONG EXIT",
         from_entry = "LONG",
         stop = longStop,
         limit = longTarget
    )
